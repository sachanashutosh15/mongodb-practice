79. Write a MongoDB query to find all the restaurants with the highest number of "A" grades.

    db.Restaurants.aggregate([
        {
            $unwind: "$grades"
        },
        {
            $match: {
                "grades.grade": "A"
            }
        },
        {
            $group: {
                _id: "$restaurant_id",
                count: {
                    $sum: 1
                }
            }
        },
        {
            $sort: {
                count: -1
            }
        },
        {
            $group: {
                _id: "$count",
                restaurantIds: {
                    $push: "$_id"
                }
            }
        },
        {
            $sort: {
                _id: -1
            }
        },
        {
            $limit: 1
